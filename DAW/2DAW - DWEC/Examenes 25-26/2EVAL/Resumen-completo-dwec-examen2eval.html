1. PROGRAMACIÓN ORIENTADA A OBJETOS (POO) - CLASES ES6

Declaración de una clase

// Sintaxis básica de una clase class Persona { // Constructor: se ejecuta al crear el objeto constructor(nombre, edad) { this.nombre = nombre; // Propiedades de instancia this.edad = edad; } // Método de instancia saludar() { console.log(`Hola, soy ${this.nombre} y tengo ${this.edad} años`); } // Otro método cumplirAnios() { this.edad++; console.log(`Ahora tengo ${this.edad} años`); } } // Crear objetos (instancias) de la clase let persona1 = new Persona("Ana", 25); let persona2 = new Persona("Luis", 30); persona1.saludar(); // "Hola, soy Ana y tengo 25 años" persona2.cumplirAnios(); // "Ahora tengo 31 años"
Conceptos clave:
  • Clase: Plantilla o molde para crear objetos
  • Constructor: Método especial que se ejecuta al crear un objeto con new
  • this: Hace referencia al objeto actual
  • Instancia: Objeto concreto creado a partir de una clase

Herencia (extends y super)

// Clase base (padre) class Animal { constructor(nombre) { this.nombre = nombre; } hacerSonido() { console.log(`${this.nombre} hace un sonido`); } } // Clase derivada (hija) que hereda de Animal class Perro extends Animal { constructor(nombre, raza) { super(nombre); // ⚠️ Llama al constructor del padre this.raza = raza; } // Método propio de Perro ladrar() { console.log(`${this.nombre} dice: ¡Guau guau!`); } // Sobrescribir método del padre hacerSonido() { console.log(`${this.nombre} ladra fuerte`); } } let miPerro = new Perro("Rex", "Pastor Alemán"); miPerro.hacerSonido(); // "Rex ladra fuerte" (método sobrescrito) miPerro.ladrar(); // "Rex dice: ¡Guau guau!"
Reglas de la herencia:
  • extends → Indica que una clase hereda de otra
  • super() → DEBE llamarse PRIMERO en el constructor hijo antes de usar this
  • La clase hija hereda todos los métodos y propiedades del padre
  • Puedes sobrescribir métodos del padre escribiendo uno con el mismo nombre

Getters y Setters

class Producto { constructor(nombre, precio) { this.nombre = nombre; this._precio = precio; // Convención: _ indica "privado" } // Getter: obtener valor (se usa como propiedad) get precio() { return `${this._precio}€`; } // Setter: establecer valor con validación set precio(nuevoPrecio) { if (nuevoPrecio < 0) { console.log("Error: El precio no puede ser negativo"); return; } this._precio = nuevoPrecio; } // Getter calculado get precioConIVA() { return this._precio * 1.21; } } let producto = new Producto("Laptop", 800); console.log(producto.precio); // "800€" (llama al getter) producto.precio = 900; // Llama al setter producto.precio = -50; // Error: El precio no puede ser negativo console.log(producto.precioConIVA); // 1089 (900 * 1.21)
Diferencias Getter vs Método normal:
  • Getter: Se usa como propiedad → producto.precio (sin paréntesis)
  • Método: Se llama con paréntesis → producto.getPrecio()
  • Setter: Se usa con asignación → producto.precio = 900

Métodos estáticos

class Matematicas { // Método estático: se llama desde la clase, NO desde instancias static sumar(a, b) { return a + b; } static restar(a, b) { return a - b; } static PI = 3.14159; // Propiedad estática } // Uso CORRECTO: llamar desde la clase console.log(Matematicas.sumar(5, 3)); // 8 console.log(Matematicas.PI); // 3.14159 // ❌ INCORRECTO: No se puede llamar desde una instancia let obj = new Matematicas(); // obj.sumar(5, 3); // ❌ Error: sumar no existe en la instancia
// Ejemplo práctico: Utilidades de validación class Validador { static validarEmail(email) { const regex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/; return regex.test(email); } static validarDNI(dni) { const regex = /^\d{8}[A-Z]$/; return regex.test(dni); } } // Uso directo sin crear objetos console.log(Validador.validarEmail("ana@example.com")); // true console.log(Validador.validarDNI("12345678Z")); // true
Métodos estáticos:
  • Se llaman directamente desde la clase: NombreClase.metodo()
  • NO tienen acceso a this (no pertenecen a ninguna instancia)
  • Útiles para funciones de utilidad que no necesitan datos de instancia
  • Ejemplo real: Math.max(), Array.isArray()

2. FORMULARIOS Y VALIDACIÓN

Eventos de formularios

let formulario = document.querySelector("#miFormulario"); let input = document.querySelector("#email"); // submit - Se dispara al enviar el formulario formulario.addEventListener("submit", (e) => { e.preventDefault(); // ⚠️ Evita que la página se recargue console.log("Formulario enviado"); }); // input - Se dispara en cada cambio (cada tecla pulsada) input.addEventListener("input", (e) => { console.log("Valor actual:", e.target.value); }); // change - Se dispara al perder el foco SI el valor cambió input.addEventListener("change", (e) => { console.log("Valor final:", e.target.value); }); // blur - Se dispara al perder el foco (siempre) input.addEventListener("blur", (e) => { console.log("El input perdió el foco"); }); // focus - Se dispara al obtener el foco input.addEventListener("focus", (e) => { console.log("El input tiene el foco"); });
Diferencias entre eventos:
Evento Cuándo se dispara Uso típico
input Cada cambio de valor (cada tecla) Validación en tiempo real
change Al perder foco SI cambió el valor Validación al terminar de escribir
blur Al perder el foco (siempre) Validación al salir del campo
submit Al enviar el formulario Validación final antes de enviar

preventDefault()

// Evitar comportamiento por defecto del navegador // En formularios: evitar recarga de página formulario.addEventListener("submit", (e) => { e.preventDefault(); // ⚠️ Evita que se envíe y recargue la página // Aquí puedes validar y procesar los datos }); // En enlaces: evitar navegación let enlace = document.querySelector("#enlace"); enlace.addEventListener("click", (e) => { e.preventDefault(); // ⚠️ Evita que el enlace navegue console.log("Enlace clickeado pero no navegó"); });
¿Cuándo usar preventDefault()?
  • En el evento submit para evitar recarga de página
  • En enlaces cuando quieres procesar el click sin navegar
  • En eventos de teclado/ratón cuando quieres anular el comportamiento por defecto

Expresiones Regulares (RegEx) para validación

// Sintaxis básica let regex = /patron/; let resultado = regex.test(cadena); // Devuelve true/false // Patrones comunes vistos en clase // Email básico let emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/; emailRegex.test("ana@example.com"); // true // Teléfono (9 dígitos) let telefonoRegex = /^\d{9}$/; telefonoRegex.test("612345678"); // true // DNI (8 dígitos + letra) let dniRegex = /^\d{8}[A-Z]$/; dniRegex.test("12345678Z"); // true // Contraseña (mínimo 8 caracteres, 1 mayúscula, 1 minúscula, 1 número) let passRegex = /^(?=.*[a-z])(?=.*[A-Z])(?=.*\d).{8,}$/; passRegex.test("MiPass123"); // true // Código postal español (5 dígitos) let cpRegex = /^\d{5}$/; cpRegex.test("50001"); // true
Símbolos de RegEx importantes:
  • ^ → Inicio de cadena
  • $ → Final de cadena
  • \d → Dígito (0-9)
  • \s → Espacio en blanco
  • [A-Z] → Cualquier letra mayúscula
  • [a-z] → Cualquier letra minúscula
  • {9} → Exactamente 9 repeticiones
  • {8,} → Mínimo 8 repeticiones
  • + → Una o más veces
  • * → Cero o más veces
  • . → Cualquier carácter
  • (?=.*[A-Z]) → Lookahead: debe contener al menos una mayúscula

try-catch-finally-throw

// Estructura básica try { // Código que puede lanzar errores let resultado = operacionPeligrosa(); console.log(resultado); } catch (error) { // Capturar y manejar el error console.log("Error capturado:", error.message); } finally { // SIEMPRE se ejecuta (opcional) console.log("Esto se ejecuta siempre"); }
// Ejemplo: Validación con throw function validarEdad(edad) { if (typeof edad !== "number") { throw new Error("La edad debe ser un número"); } if (edad < 0) { throw new Error("La edad no puede ser negativa"); } if (edad < 18) { throw new Error("Debes ser mayor de edad"); } return true; } try { validarEdad(15); } catch (error) { console.log(error.message); // "Debes ser mayor de edad" }
// Ejemplo: Parsear JSON con try-catch function cargarDatos() { let datosJSON = localStorage.getItem("usuarios"); try { let usuarios = JSON.parse(datosJSON); return usuarios || []; } catch (error) { console.log("Error al parsear JSON:", error.message); return []; // Devolver array vacío si hay error } }
Conceptos clave de try-catch:
  • try → Bloque de código que puede fallar
  • catch(error) → Captura el error y lo maneja
  • finally → Se ejecuta SIEMPRE (con error o sin error)
  • throw → Lanza un error manualmente
  • error.message → Mensaje del error

Promesas (Promises)

// Una promesa tiene 3 estados: // - Pending (pendiente): estado inicial // - Fulfilled (cumplida): operación completada con éxito // - Rejected (rechazada): operación falló // Crear una promesa let miPromesa = new Promise((resolve, reject) => { let exito = true; if (exito) { resolve("Operación exitosa"); // Cumplir la promesa } else { reject("Operación fallida"); // Rechazar la promesa } }); // Consumir una promesa miPromesa .then((resultado) => { console.log(resultado); // Se ejecuta si resolve() }) .catch((error) => { console.log(error); // Se ejecuta si reject() }) .finally(() => { console.log("Promesa finalizada"); // Siempre se ejecuta });
// Ejemplo práctico: Simulación de petición function obtenerDatos() { return new Promise((resolve, reject) => { setTimeout(() => { let exito = true; if (exito) { resolve({nombre: "Ana", edad: 25}); } else { reject("Error al obtener datos"); } }, 2000); }); } obtenerDatos() .then((datos) => { console.log("Datos recibidos:", datos); }) .catch((error) => { console.log("Error:", error); });
// Encadenar promesas function paso1() { return new Promise((resolve) => { setTimeout(() => { console.log("Paso 1 completado"); resolve(1); }, 1000); }); } function paso2(resultado) { return new Promise((resolve) => { setTimeout(() => { console.log("Paso 2 completado"); resolve(resultado + 1); }, 1000); }); } function paso3(resultado) { return new Promise((resolve) => { setTimeout(() => { console.log("Paso 3 completado"); resolve(resultado + 1); }, 1000); }); } // Encadenar paso1() .then((resultado1) => paso2(resultado1)) .then((resultado2) => paso3(resultado2)) .then((resultadoFinal) => { console.log("Resultado final:", resultadoFinal); // 3 }) .catch((error) => { console.log("Error en algún paso:", error); });
Promesas: Conceptos clave
  • .then() → Se ejecuta cuando la promesa se cumple (resolve)
  • .catch() → Se ejecuta cuando la promesa se rechaza (reject)
  • .finally() → Se ejecuta SIEMPRE al finalizar
  • Las promesas se pueden encadenar para operaciones secuenciales
  • Ventaja sobre callbacks: código más legible y fácil de mantener

Async / Await

// async/await es una forma más limpia de trabajar con promesas // Función asíncrona (devuelve una promesa automáticamente) async function obtenerDatos() { return "Datos obtenidos"; } // Es equivalente a: function obtenerDatos() { return Promise.resolve("Datos obtenidos"); } // Consumir con .then() obtenerDatos().then((datos) => console.log(datos));
// await: espera a que una promesa se resuelva function esperarDosSegundos() { return new Promise((resolve) => { setTimeout(() => { resolve("Han pasado 2 segundos"); }, 2000); }); } // ❌ Sin await (promesa pendiente) function obtenerMensaje() { let mensaje = esperarDosSegundos(); console.log(mensaje); // Promise {} } // ✅ Con await (espera el resultado) async function obtenerMensaje() { let mensaje = await esperarDosSegundos(); console.log(mensaje); // "Han pasado 2 segundos" } obtenerMensaje();
// Ejemplo completo con try-catch async function cargarDatosUsuario(id) { try { console.log("Cargando usuario..."); let usuario = await obtenerUsuario(id); console.log("Cargando posts..."); let posts = await obtenerPosts(usuario.id); console.log("Cargando comentarios..."); let comentarios = await obtenerComentarios(posts[0].id); return {usuario, posts, comentarios}; } catch (error) { console.log("Error al cargar datos:", error); throw error; // Re-lanzar el error } } cargarDatosUsuario(1) .then((datos) => console.log(datos)) .catch((error) => console.log("Error:", error));
async / await: Reglas importantes
  • async antes de la función → La convierte en asíncrona
  • await solo funciona dentro de funciones async
  • await pausa la ejecución hasta que la promesa se resuelva
  • SIEMPRE usa try-catch para manejar errores
  • Ventaja: código asíncrono que parece síncrono (más legible)

Manejo de errores y Status Codes

async function obtenerUsuario(id) { try { let response = await fetch(`https://api.example.com/usuarios/${id}`); // Verificar si la respuesta fue exitosa if (!response.ok) { throw new Error(`Error HTTP: ${response.status}`); } let usuario = await response.json(); console.log(usuario); } catch (error) { console.log("Error:", error.message); } }
// Manejo detallado según status code async function obtenerUsuario(id) { try { let response = await fetch(`https://api.example.com/usuarios/${id}`); if (response.status === 404) { console.log("Usuario no encontrado"); return null; } if (response.status === 401) { console.log("No autorizado - necesitas iniciar sesión"); return null; } if (!response.ok) { throw new Error(`Error: ${response.status}`); } let usuario = await response.json(); return usuario; } catch (error) { console.log("Error de red:", error.message); return null; } }

return null; } } // READ - Obtener todos los usuarios async function obtenerUsuarios() { try { let response = await fetch(API_URL); if (!response.ok) { throw new Error(`Error: ${response.status}`); } let usuarios = await response.json(); console.log("Usuarios:", usuarios); return usuarios; } catch (error) { console.log("Error al obtener usuarios:", error.message); return []; } } // READ - Obtener un usuario por ID async function obtenerUsuario(id) { try { let response = await fetch(`${API_URL}/${id}`); if (response.status === 404) { console.log("Usuario no encontrado"); return null; } if (!response.ok) { throw new Error(`Error: ${response.status}`); } let usuario = await response.json(); console.log("Usuario:", usuario); return usuario; } catch (error) { console.log("Error al obtener usuario:", error.message); return null; } } // UPDATE - Actualizar usuario async function actualizarUsuario(id, datosActualizados) { try { let response = await fetch(`${API_URL}/${id}`, { method: "PUT", headers: { "Content-Type": "application/json" }, body: JSON.stringify(datosActualizados) }); if (!response.ok) { throw new Error(`Error: ${response.status}`); } let usuarioActualizado = await response.json(); console.log("Usuario actualizado:", usuarioActualizado); return usuarioActualizado; } catch (error) { console.log("Error al actualizar usuario:", error.message); return null; } } // DELETE - Eliminar usuario async function eliminarUsuario(id) { try { let response = await fetch(`${API_URL}/${id}`, { method: "DELETE" }); if (!response.ok) { throw new Error(`Error: ${response.status}`); } console.log(`Usuario ${id} eliminado`); return true; } catch (error) { console.log("Error al eliminar usuario:", error.message); return false; } } // Uso crearUsuario({name: "Ana García", email: "ana@example.com"}); obtenerUsuarios(); obtenerUsuario(1); actualizarUsuario(1, {name: "Ana López"}); eliminarUsuario(1);
